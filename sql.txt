-- Create the primary database
CREATE DATABASE IF NOT EXISTS org_accreditation_db;
USE org_accreditation_db;

-- SINGLE-VALUED TABLES 

-- 1. Org_Details_Table
CREATE TABLE org_details (
    org_id VARCHAR(255) NOT NULL, 
    org_name VARCHAR(50) NOT NULL, 
    office_address VARCHAR(60),
    date_organized DATE, 
    contact_number VARCHAR(11), 
    org_level VARCHAR(22) NOT NULL,  
    linkages_membership VARCHAR(13) NOT NULL, 
    sector_served VARCHAR(42) NOT NULL, 
    PRIMARY KEY (org_id)
);

-- 2. Accreditation_Table
CREATE TABLE accreditation (
    org_id VARCHAR(255) NOT NULL, 
    accreditation_year INT(4) NOT NULL, 
    renewal_status VARCHAR(21) NOT NULL, 
    male_members INT(6) DEFAULT 0, 
    female_members INT(6) DEFAULT 0, 
    total_members INT(6) DEFAULT 0, 
    unregistered_voter_count INT(6) DEFAULT 0, 
    registered_voter_count INT(6) DEFAULT 0, 
    PRIMARY KEY (org_id, accreditation_year),
    FOREIGN KEY (org_id) REFERENCES org_details(org_id) 
        ON DELETE CASCADE 
        ON UPDATE CASCADE
);

-- MULTI-VALUED TABLES (Dependent Tables)

-- 3. Agency_Table
CREATE TABLE registering_agency (
    agency_id INT AUTO_INCREMENT NOT NULL, 
    org_id VARCHAR(255) NOT NULL, 
    accreditation_year INT(4) NOT NULL, 
    registering_agency VARCHAR(45) NOT NULL, 
    PRIMARY KEY (agency_id),
    FOREIGN KEY (org_id, accreditation_year) REFERENCES accreditation(org_id, accreditation_year)
        ON DELETE CASCADE 
        ON UPDATE CASCADE
);

-- 4. Purpose_Table
CREATE TABLE purpose (
    purpose_id INT AUTO_INCREMENT NOT NULL, 
    org_id VARCHAR(255) NOT NULL, 
    accreditation_year INT(4) NOT NULL, 
    purpose VARCHAR(15) NOT NULL, 
    PRIMARY KEY (purpose_id),
    FOREIGN KEY (org_id, accreditation_year) REFERENCES accreditation(org_id, accreditation_year)
        ON DELETE CASCADE 
        ON UPDATE CASCADE
);

-- 5. Services_Facilities_Table
CREATE TABLE services_facilities (
    services_id INT AUTO_INCREMENT NOT NULL,  
    org_id VARCHAR(255) NOT NULL, 
    accreditation_year INT(4) NOT NULL, 
    services_facilities VARCHAR(22) NOT NULL, 
    PRIMARY KEY (services_id),
    FOREIGN KEY (org_id, accreditation_year) REFERENCES accreditation(org_id, accreditation_year)
        ON DELETE CASCADE 
        ON UPDATE CASCADE
);

-- 6. Finance_Table
CREATE TABLE finance (
    finance_id INT AUTO_INCREMENT NOT NULL, 
    org_id VARCHAR(255) NOT NULL,  
    accreditation_year INT(4) NOT NULL,  
    financing_source VARCHAR(16) NOT NULL,  
    PRIMARY KEY (finance_id),
    FOREIGN KEY (org_id, accreditation_year) REFERENCES accreditation(org_id, accreditation_year)
        ON DELETE CASCADE 
        ON UPDATE CASCADE
);

-- 7. Local_Body_Table
CREATE TABLE local_body_priority (
    local_body_id INT AUTO_INCREMENT NOT NULL, 
    org_id VARCHAR(255) NOT NULL, 
    accreditation_year INT(4) NOT NULL, 
    local_body_priority VARCHAR(62) NOT NULL, 
    PRIMARY KEY (local_body_id),
    FOREIGN KEY (org_id, accreditation_year) REFERENCES accreditation(org_id, accreditation_year)
        ON DELETE CASCADE 
        ON UPDATE CASCADE
);
